`timescale 1ns/1ps
module tb_mono_config_ack_pipeline;
    reg clk=0;
    always #4 clk=!clk;
    reg rst_n=0;
    reg [39:0] command_data=0;
    reg command_valid=0;
    wire command_ready;
    wire [39:0] response_data;
    wire response_valid;
    reg response_ready=0;
    reg pipeline_idle=1,frame_accept=0;
    reg [31:0] frame_sequence=0;
    wire source_ready,pause_new_jobs,pixel_initialized,config_fault;
    wire [1183:0] m_config_data,active_config;
    wire m_config_valid;
    reg m_config_ready=0;
    reg [1183:0] s_config_ack_data=0;
    reg s_config_ack_valid=0,s_config_ack_ok=1;
    wire s_config_ack_ready;
    wire [31:0] config_stale_ack_count,mode_epoch;
    wire [1:0] active_mode,active_product_mode;
    wire [7:0] active_quality,minimum_quality;
    wire adaptive_quality,active_schema2,active_legacy_gaussian;
    wire [24:0] frame_budget;
    wire [15:0] restart_mcus;
    wire [31:0] source_session;
    wire [5:0] active_processing_enables;
    mono_config_engine_v2 #(.V2_SUPPORTED_MODES(3'b101),.V2_SUPPORTED_ENABLES(6'h3f)) dut(.*);
    integer checks=0,ack_flips=0,limit,mutation_bit,phase=0,reset_stage,ack_transfers=0,ack_stalls=0;
    reg [1183:0] boot_plan,commit_plan,changed_plan;
    reg [1023:0] extension;
    reg [39:0] held_response;
    always @(posedge clk)begin
        if(rst_n&&s_config_ack_valid)begin
            if(s_config_ack_ready)ack_transfers++;
            else ack_stalls++;
        end
    end
    task check(input bit condition,input string reason);
        begin checks++;if(!condition)$fatal(1,"ACK pipeline phase=%0d mutation=%0d: %s",phase,mutation_bit,reason);end
    endtask
    task flit(input [39:0] value);
        begin
            @(negedge clk);command_data=value;command_valid=1;limit=0;
            do begin @(posedge clk);limit++;check(limit<100,"command ready bounded");end while(!command_ready);
            @(negedge clk);command_valid=0;
        end
    endtask
    task packet(input [1:0] kind);
        integer word_index;
        reg [31:0] value;
        begin
            flit({1'b1,kind==3,kind,kind!=3,3'd0,32'h12345678});
            if(kind!=3)for(word_index=0;word_index<41;word_index++)begin
                case(word_index)
                    0:value=17;1:value=32'h12345678;2:value=0;3:value=70;
                    4:value=1;5:value=7;6:value=12345;7:value=17;8:value=1;
                    default:value=extension[(word_index-9)*32+:32];
                endcase
                flit({1'b0,word_index==40,6'(word_index),value});
            end
        end
    endtask
    task receive_reply(input [1:0] kind,input [31:0] sequence_value,epoch_value,last_value);
        integer word_index;
        reg [31:0] expected;
        begin
            for(word_index=0;word_index<4;word_index++)begin
                limit=0;
                while(!response_valid)begin @(negedge clk);limit++;check(limit<100,"response bounded");end
                held_response=response_data;
                repeat(5)begin @(negedge clk);check(response_valid&&response_data===held_response,"held response stable");end
                case(word_index)
                    0:expected=32'h12345678;1:expected=sequence_value;2:expected=epoch_value;3:expected=last_value;
                endcase
                check(response_data[31:0]===expected,"response field identity");
                check(response_data[39:32]==(word_index==0 ? {2'b10,kind,4'd0} : {1'b0,word_index==3,6'(word_index-1)}),"response kind/index/last");
                response_ready=1;@(posedge clk);@(negedge clk);response_ready=0;
            end
        end
    endtask
    task send_ack(input [1183:0] value,input bit ok);
        begin
            @(negedge clk);s_config_ack_data=value;s_config_ack_ok=ok;s_config_ack_valid=1;limit=0;
            do begin @(posedge clk);limit++;check(limit<100,"ACK ready bounded");end while(!s_config_ack_ready);
            @(negedge clk);s_config_ack_valid=0;s_config_ack_data=~value;s_config_ack_ok=!ok;
            check(!s_config_ack_ready,"comparison backpressures next ACK");
            limit=0;
            while(!s_config_ack_ready)begin @(negedge clk);limit++;check(limit<100,"comparison completion bounded");end
        end
    endtask
    task take_plan(output reg [1183:0] value);
        begin
            limit=0;
            while(!m_config_valid)begin @(negedge clk);limit++;check(limit<100,"plan bounded");end
            value=m_config_data;
            repeat(5)begin @(negedge clk);check(m_config_valid&&m_config_data===value,"held plan stable");end
            m_config_ready=1;@(posedge clk);@(negedge clk);m_config_ready=0;
        end
    endtask
    task reset_engine;
        begin
            @(negedge clk);rst_n=0;command_valid=0;s_config_ack_valid=0;frame_accept=0;response_ready=0;
            repeat(3)@(negedge clk);rst_n=1;
        end
    endtask
    initial begin
        extension=0;extension[7:0]=2;extension[9:8]=2;extension[15:10]=6'h3f;
        extension[66:64]=4;extension[76:67]=50;extension[86:77]=50;extension[108:105]=15;
        extension[191:128]=64'h800000019abcdef0;extension[223:192]=32'h80001001;
        extension[255:224]=32'hf0f1f2f3;extension[383:352]=32'hfedca987;
        reset_engine();phase=1;packet(3);
        wait(m_config_valid);boot_plan=m_config_data;
        send_ack(boot_plan,1);
        check(config_stale_ack_count==1&&!pixel_initialized&&!response_valid,"premature ACK before plan fire rejected");
        take_plan(boot_plan);
        for(mutation_bit=0;mutation_bit<1184;mutation_bit++)begin
            changed_plan=boot_plan;changed_plan[mutation_bit]=!changed_plan[mutation_bit];send_ack(changed_plan,1);ack_flips++;
            check(!pixel_initialized&&!source_ready&&pause_new_jobs&&!response_valid&&!config_fault,"BOOT all bits checked before admission");
        end
        send_ack(boot_plan,1);receive_reply(0,0,0,0);
        check(pixel_initialized&&active_config===boot_plan,"BOOT initializes exact defaults");
        send_ack(boot_plan,0);
        check(!config_fault&&config_stale_ack_count==1186,"duplicate negative ACK cannot fault initialized transaction");
        phase=2;packet(0);receive_reply(1,17,0,0);packet(1);take_plan(commit_plan);
        for(mutation_bit=0;mutation_bit<1184;mutation_bit++)begin
            changed_plan=commit_plan;changed_plan[mutation_bit]=!changed_plan[mutation_bit];send_ack(changed_plan,1);ack_flips++;
            check(!source_ready&&pause_new_jobs&&!response_valid&&mode_epoch==0&&active_config===boot_plan,"COMMIT all bits checked before frame release");
        end
        phase=3;
        @(negedge clk);s_config_ack_data=commit_plan;s_config_ack_ok=1;s_config_ack_valid=1;frame_accept=1;frame_sequence=91;
        @(posedge clk);check(s_config_ack_ready&&!source_ready&&mode_epoch==0,"ACK/SOF receive edge blocked");
        @(negedge clk);s_config_ack_valid=0;s_config_ack_data=0;s_config_ack_ok=0;
        @(posedge clk);check(!source_ready&&mode_epoch==0&&!response_valid,"ACK/SOF reduction edge blocked");
        @(posedge clk);check(!source_ready&&mode_epoch==0&&!response_valid,"ACK/SOF result retirement edge blocked");
        @(negedge clk);frame_accept=0;#1;
        check(source_ready&&!pause_new_jobs&&mode_epoch==0&&active_config===boot_plan,"positive ACK releases without inventing APPLIED");
        pipeline_idle=0;frame_accept=1;frame_sequence=92;repeat(3)@(negedge clk);frame_accept=0;
        check(mode_epoch==0&&!response_valid,"busy pipeline cannot manufacture apply frame");
        pipeline_idle=1;repeat(3)@(negedge clk);frame_sequence=99;frame_accept=1;
        @(posedge clk);@(negedge clk);frame_accept=0;
        check(mode_epoch==1&&active_config===commit_plan,"actual accepted frame publishes exact prepared epoch");
        receive_reply(2,17,1,99);
        phase=4;
        for(reset_stage=0;reset_stage<2;reset_stage++)begin
            reset_engine();packet(3);take_plan(boot_plan);
            @(negedge clk);s_config_ack_data=boot_plan;s_config_ack_ok=1;s_config_ack_valid=1;
            @(posedge clk);@(negedge clk);s_config_ack_valid=0;
            repeat(reset_stage)@(negedge clk);rst_n=0;
            repeat(3)@(negedge clk);rst_n=1;repeat(7)@(negedge clk);
            check(!pixel_initialized&&!response_valid&&!config_fault&&config_stale_ack_count==0,"reset cancels each comparison stage");
        end
        packet(3);take_plan(boot_plan);send_ack(boot_plan,0);
        check(config_fault&&!source_ready&&pause_new_jobs&&!pixel_initialized,"matching negative ACK faults exact transaction");
        receive_reply(3,0,6,0);
        phase=5;reset_engine();packet(3);take_plan(boot_plan);
        ack_transfers=0;ack_stalls=0;
        @(negedge clk);s_config_ack_data=boot_plan;s_config_ack_ok=1;s_config_ack_valid=1;
        repeat(9)@(negedge clk);
        s_config_ack_valid=0;
        check(ack_transfers==3&&ack_stalls==6,"held ACK obeys busy backpressure and recovers");
        check(pixel_initialized&&!config_fault&&config_stale_ack_count==2,"duplicate ACKs counted once per actual handshake");
        receive_reply(0,0,0,0);
        check(!response_valid,"held duplicates never emit extra BOOT response");
        $display("RESULT: PASS mono_config_ack_pipeline checks=%0d ack_bit_mutations=%0d held_ack_transfers=%0d held_ack_stalls=%0d reset_stages=2",checks,ack_flips,ack_transfers,ack_stalls);$finish;
    end
    initial begin #2000000;$fatal(1,"ACK pipeline watchdog");end
endmodule
